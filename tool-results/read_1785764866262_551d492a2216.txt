   785→function Stepper({
   786→  steps,
   787→  active,
   788→}: {
   789→  steps: { label: string; href: string }[];
   790→  active: number;
   791→}) {
   792→  return (
   793→    <ol className="flex items-center justify-between gap-1 rounded-2xl bg-white/70 p-3 shadow-[0_8px_30px_rgba(61,0,46,0.08)] backdrop-blur-sm">
   794→      {steps.map((step, i) => {
   795→        const isDone = i < active;
   796→        const isActive = i === active;
   797→        return (
   798→          <li
   799→            key={step.label}
   800→            className="flex flex-1 flex-col items-center gap-1.5 text-center"
   801→          >
   802→            <div className="flex w-full items-center">
   803→              {i > 0 && (
   804→                <span
   805→                  className={`h-0.5 flex-1 ${
   806→                    isDone || isActive ? "bg-[#F10897]" : "bg-maroon/15"
   807→                  }`}
   808→                />
   809→              )}
   810→              <span
   811→                className={`flex h-7 w-7 shrink-0 items-center justify-center rounded-full font-sans text-[11px] font-bold transition-colors ${
   812→                  isActive
   813→                    ? "bg-[#F10897] text-white shadow-[0_4px_12px_rgba(241,8,151,0.4)]"
   814→                    : isDone
   815→                      ? "bg-[#4E0030] text-white"
   816→                      : "bg-maroon/10 text-maroon/60"
   817→                }`}
   818→              >
   819→                {isDone ? <Check className="h-3.5 w-3.5" strokeWidth={2.5} /> : i + 1}
   820→              </span>
   821→              {i < steps.length - 1 && (
   822→                <span
   823→                  className={`h-0.5 flex-1 ${
   824→                    i < active ? "bg-[#F10897]" : "bg-maroon/15"
   825→                  }`}
   826→                />
   827→              )}
   828→            </div>
   829→            <span
   830→              className={`font-sans text-[10px] font-bold uppercase tracking-[0.1em] ${
   831→                isActive
   832→                  ? "text-[#F10897]"
   833→                  : isDone
   834→                    ? "text-maroon"
   835→                    : "text-maroon/50"
   836→              }`}
   837→            >
   838→              {step.label}
   839→            </span>
   840→          </li>
   841→        );
   842→      })}
   843→    </ol>
   844→  );
   845→}
