    89→    89→/* Seeded hash → deterministic redemption code per recipient.
    90→    90→   Produces XXXX-XXXX-XXXX-XXXX (alphanumeric, uppercase). */
    91→    91→function seededRedemptionCode(seed: string): string {
    92→    92→  let h1 = 0xdeadbeef ^ 0;
    93→    93→  let h2 = 0x41c6ce57 ^ 0;
    94→    94→  for (let i = 0; i < seed.length; i++) {
    95→    95→    const ch = seed.charCodeAt(i);
    96→    96→    h1 = Math.imul(h1 ^ ch, 2654435761);
    97→    97→    h2 = Math.imul(h2 ^ ch, 1597334677);
    98→    98→  }
    99→    99→  h1 = Math.imul(h1 ^ (h1 >>> 16), 2246822507) ^ Math.imul(h2 ^ (h2 >>> 13), 3266489909);
   100→   100→  h2 = Math.imul(h2 ^ (h2 >>> 16), 2246822507) ^ Math.imul(h1 ^ (h1 >>> 13), 3266489909);
   101→   101→  const alphabet = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789"; // no ambiguous chars
   102→   102→  const total = 16;
   103→   103→  const chunks: string[] = [];
   104→   104→  let combined = (h1 >>> 0).toString(2).padStart(32, "0") + (h2 >>> 0).toString(2).padStart(32, "0");
   105→   105→  // Ensure deterministic length by mixing
   106→   106→  while (combined.length < total * 6) {
   107→   107→    combined += combined;
   108→   108→  }
   109→   109→  for (let i = 0; i < total; i++) {
   110→   110→    const bits = combined.slice(i * 6, i * 6 + 6);
   111→   111→    const idx = parseInt(bits, 2) % alphabet.length;
   112→   112→    chunks.push(alphabet[idx]);
   113→   113→  }
   114→   114→  const s = chunks.join("");
   115→   115→  return `${s.slice(0, 4)}-${s.slice(4, 8)}-${s.slice(8, 12)}-${s.slice(12, 16)}`;
   116→   116→}
