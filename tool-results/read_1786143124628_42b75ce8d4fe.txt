   918→   918→function Stepper({
   919→   919→  steps,
   920→   920→  active,
   921→   921→}: {
   922→   922→  steps: { label: string; href: string }[];
   923→   923→  active: number;
   924→   924→}) {

   928→   928→        const isDone = i < active;
   929→   929→        const isActive = i === active;

   932→   932→            key={step.label}
   933→   933→            className="relative flex flex-1 flex-col items-center gap-2 text-center"

   941→   941→                aria-hidden
   942→   942→              />

   968→   968→          </li>

   972→   972→  );
   973→   973→}
