    88→/* Seeded hash → deterministic redemption code per recipient.
    89→   Produces XXXX-XXXX-XXXX-XXXX (alphanumeric, uppercase). */
    90→function seededRedemptionCode(seed: string): string {
    91→  let h1 = 0xdeadbeef ^ 0;
    92→  let h2 = 0x41c6ce57 ^ 0;
    93→  for (let i = 0; i < seed.length; i++) {
    94→    const ch = seed.charCodeAt(i);
    95→    h1 = Math.imul(h1 ^ ch, 2654435761);
    96→    h2 = Math.imul(h2 ^ ch, 1597334677);
    97→  }
    98→  h1 = Math.imul(h1 ^ (h1 >>> 16), 2246822507) ^ Math.imul(h2 ^ (h2 >>> 13), 3266489909);
    99→  h2 = Math.imul(h2 ^ (h2 >>> 16), 2246822507) ^ Math.imul(h1 ^ (h1 >>> 13), 3266489909);
   100→  const alphabet = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789"; // no ambiguous chars
   101→  const total = 16;
   102→  const chunks: string[] = [];
   103→  let combined = (h1 >>> 0).toString(2).padStart(32, "0") + (h2 >>> 0).toString(2).padStart(32, "0");
   104→  // Ensure deterministic length by mixing
   105→  while (combined.length < total * 6) {
   106→    combined += combined;
   107→  }
   108→  for (let i = 0; i < total; i++) {
   109→    const bits = combined.slice(i * 6, i * 6 + 6);
   110→    const idx = parseInt(bits, 2) % alphabet.length;
   111→    chunks.push(alphabet[idx]);
   112→  }
   113→  const s = chunks.join("");
   114→  return `${s.slice(0, 4)}-${s.slice(4, 8)}-${s.slice(8, 12)}-${s.slice(12, 16)}`;
   115→}
