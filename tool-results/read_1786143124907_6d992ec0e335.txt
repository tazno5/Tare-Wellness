   871→   871→function Stepper({
   872→   872→  steps,
   873→   873→  active,
   874→   874→}: {
   875→   875→  steps: { label: string; href: string }[];
   876→   876→  active: number;

   900→   900→                  : isDone

   903→   903→              }`}
   904→   904→            >
