   871→function Stepper({
   872→  steps,
   873→  active,
   874→}: {
   875→  steps: { label: string; href: string }[];
   876→  active: number;
   877→}) {
   878→  return (

   880→      {steps.map((step, i) => {

   883→        return (
   884→          <li

   886→            className="flex flex-1 flex-col items-center gap-1.5 text-center"

   892→                    isDone || isActive ? "bg-[#F10897]" : "bg-maroon/15"

   899→                    ? "bg-[#F10897] text-white shadow-[0_4px_12px_rgba(241,8,151,0.4)]"
   900→                    : isDone
   901→                      ? "bg-[#4E0030] text-white"
   902→                      : "bg-maroon/10 text-maroon/60"
   903→                }`}
   904→              >
   905→                {isDone ? <Check className="h-3.5 w-3.5" strokeWidth={2.5} /> : i + 1}
   906→              </span>
   907→              {i < steps.length - 1 && (
   908→                <span
   909→                  className={`h-0.5 flex-1 ${
   910→                    i < active ? "bg-[#F10897]" : "bg-maroon/15"
   911→                  }`}

   916→              className={`font-sans text-[10px] font-bold uppercase tracking-[0.1em] ${

   918→                  ? "text-[#F10897]"
   919→                  : isDone

   923→            >
   924→              {step.label}
