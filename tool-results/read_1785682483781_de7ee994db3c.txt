   730→function Stepper({
   731→  steps,
   732→  active,
   733→}: {
   734→  steps: { label: string; href: string }[];
   735→  active: number;
   736→}) {
   737→  return (
   738→    <ol className="flex items-center justify-between gap-1 rounded-2xl bg-white/70 p-3 shadow-[0_8px_30px_rgba(61,0,46,0.08)] backdrop-blur-sm">
   739→      {steps.map((step, i) => {
   740→        const isDone = i < active;
   741→        const isActive = i === active;
   742→        return (
   743→          <li
   744→            key={step.label}
   745→            className="flex flex-1 flex-col items-center gap-1.5 text-center"
   746→          >
   747→            <div className="flex w-full items-center">
   748→              {i > 0 && (
   749→                <span
   750→                  className={`h-0.5 flex-1 ${
   751→                    isDone || isActive ? "bg-[#F10897]" : "bg-maroon/15"
   752→                  }`}
   753→                />
   754→              )}
   755→              <span
   756→                className={`flex h-7 w-7 shrink-0 items-center justify-center rounded-full font-sans text-[11px] font-bold transition-colors ${
   757→                  isActive
   758→                    ? "bg-[#F10897] text-white shadow-[0_4px_12px_rgba(241,8,151,0.4)]"
   759→                    : isDone
   760→                      ? "bg-[#4E0030] text-white"
   761→                      : "bg-maroon/10 text-maroon/60"
   762→                }`}
   763→              >
   764→                {isDone ? <Check className="h-3.5 w-3.5" strokeWidth={2.5} /> : i + 1}
   765→              </span>
   766→              {i < steps.length - 1 && (
   767→                <span
   768→                  className={`h-0.5 flex-1 ${
   769→                    i < active ? "bg-[#F10897]" : "bg-maroon/15"
   770→                  }`}
   771→                />
   772→              )}
   773→            </div>
   774→            <span
   775→              className={`font-sans text-[10px] font-bold uppercase tracking-[0.1em] ${
   776→                isActive
   777→                  ? "text-[#F10897]"
   778→                  : isDone
   779→                    ? "text-maroon"
   780→                    : "text-maroon/50"
   781→              }`}
   782→            >
   783→              {step.label}
   784→            </span>
   785→          </li>
   786→        );
   787→      })}
   788→    </ol>
   789→  );
